152 Type end_value_;
153
154 Getter<Type> getter_;
155 Setter<Type> setter_;
156 protected:
157 void hadSetPlay() override {
158 if (getter_)
159 start_value_ = getter_();
160 }
161 void playingUpdate(const TimePoint& now) override {
162 using namespace std::chrono;
163 finished_ = duration_cast<milliseconds>(now - start_time_);
164 auto elapsed_ns = duration_cast<nanoseconds>(finished_).count();
165 auto progress = elapsed_ns * 1e-6f / duration_ms_;
166 if (progress > 1.f) {
167 setStop();
168 onFinished();
169 start_time_ = now;
170 return;
171 }
172 setter_(start_value_ + (end_value_ - start_value_) * trans_func_(progress));
173 onUpdated();
174 }
175 public:
178 Animation(const AnimationParams<Type>& params)
179 : setter_(params.setter), end_value_(params.target),
180 AnimationAbstract(TimePoint(), params.duration, params.trans_func) {
181 setInitial(params.initial);
182 }
187 void setInitial(const Initial<Type>& initial) {
188 auto ptr = std::get_if<Type>(&initial);
189 ptr ? start_value_ = *ptr, getter_ = Getter<Type>()
190 : getter_ = std::get<Getter<Type>>(initial);
191 }
194 void setTarget(const Type& target) { end_value_ = target; }
